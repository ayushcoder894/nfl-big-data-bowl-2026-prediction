```python
def select_feature_subset(X, feature_names, fraction, seed):
    """Select random subset of features for diversity."""
    np.random.seed(seed)
    n_features = int(len(feature_names) * fraction)
    selected_indices = np.random.choice(
        len(feature_names), 
        size=n_features, 
        replace=False
    )
    selected_features = [feature_names[i] for i in sorted(selected_indices)]
    return X[:, selected_indices], selected_features
```

```python
def bootstrap_sample(X, y, weights, fraction, seed):
    """Create bootstrap sample with weights."""
    np.random.seed(seed)
    n_samples = int(X.shape[0] * fraction)
    indices = np.random.choice(
        X.shape[0], 
        size=n_samples, 
        replace=True
    )
    return X[indices], y[indices], weights[indices], indices

# For each model:
```

```python
def select_feature_subset(X, feature_fraction, seed):
    """Select random subset of features for model diversity."""
    if feature_fraction >= 1.0:
        return X, list(range(X.shape[1]))
    
    np.random.seed(seed)
    n_features = int(X.shape[1] * feature_fraction)
    n_features = max(n_features, int(X.shape[1] * 0.5))  # Use at least 50%
    
    selected_indices = np.random.choice(
        X.shape[1], 
        size=n_features, 
        replace=False
    )
    selected_indices = np.sort(selected_indices)
    
    return X[:, selected_indices], list(selected_indices)

def bootstrap_sample(X, y, weights, sample_fraction, seed):
    """Create bootstrap sample for model diversity."""
    if sample_fraction >= 1.0:
        return X, y, weights
    
    np.random.seed(seed)
    n_samples = int(X.shape[0] * sample_fraction)
    
    indices = np.random.choice(
        X.shape[0], 
        size=n_samples, 
        replace=True
    )
    
    return X[indices], y[indices], weights[indices]
```

--------------------------------
Risk: Important features might be excluded

- Use at least 70% of features per model (ensures coverage)
- Linear model (HUBER) uses 100% of features
- Meta-learner will compensate for missing features

-------------------------

- Minimal computational overhead
